def audit(event, args):
    forbidden = event.startswith("sqlite3.connect") or event in {"socket.connect", "socket.connect_ex", "socket.getaddrinfo", "socket.bind", "subprocess.Popen", "os.system"}
    if event == "import" and str(args[0]).split(".")[0] == "keyring":
        forbidden = True
    if event == "open":
        name = str(args[0]).replace("\\", "/").lower()
        forbidden = forbidden or name.endswith((".db", ".sqlite", ".sqlite3", ".key")) or "/.codex/" in name or "/.jippeel/" in name or name.endswith("auth.json")
        mode = args[1]
        flags = args[2]
        writing = (isinstance(mode, str) and any(c in mode for c in "wax+")) or (isinstance(flags, int) and bool(flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC)))
        if writing:
            try:
                pathlib.Path(args[0]).resolve().relative_to(RUN)
            except (ValueError, TypeError):
                forbidden = True
    if forbidden:
        BLOCKED.append(event)
        raise RuntimeError("PROBE_GUARD: " + event)

def outcome(call):
    try:
        return {"output": call()}
    except (ValueError, TypeError) as exc:
        return {"exception": type(exc).__name__, "message": str(exc)}

    canon_results.append({"id": name, "input_raw": raw, "expected": expected + " (invalid schema must raise ValueError for repair)", **result, "verdict": "PASS" if meets else "RED"})

# All awaits finish inline in the deterministic fake; no event loop/network is created.
def finish_inline(coroutine):
    try:
        coroutine.send(None)
    except StopIteration as exc:
        return exc.value
    finally:
        coroutine.close()
    raise RuntimeError("PROBE_INFRA: unexpected suspended await")
